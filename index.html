كود مانيم: دراسة دالة تربيعية واتجاه تغيراتها

هذا الكود من مشروع حقيقي على GitHub يوضح دراسة الدالة التربيعية وتحليل اتجاه تغيراتها

# دراسة الدالة التربيعية وتحليل اتجاه تغيراتها
# مشروع على GitHub: https://github.com/beneddine-dotcom/manim_project/issues/24

from manim import *

class QuadraticFunctionAnalysis(Scene):
    def construct(self):
        # إنشاء نظام الإحداثيات
        axes = Axes(
            x_range=[-3, 3, 1],
            y_range=[-1, 5, 1],
            x_length=7,
            y_length=5,
            axis_config={"color": BLUE}
        )

        # الدالة التربيعية: f(x) = x² - 2x + 1
        quadratic_func = axes.plot(
            lambda x: x**2 - 2*x + 1,
            color=GREEN,
            stroke_width=4
        )

        # إيجاد رأس المنحنى (1, 0)
        vertex = Dot(axes.c2p(1, 0), color=RED, radius=0.1)
        vertex_label = MathTex("(1, 0)").next_to(vertex, DOWN)

        # دراسة اتجاه التغيرات
        decreasing_text = Text("تزايدية", font="Arial", font_size=24, color=YELLOW)
            .next_to(axes.c2p(-2, 3), LEFT)
        increasing_text = Text("تناقصية", font="Arial", font_size=24, color=ORANGE)
            .next_to(axes.c2p(2, 3), RIGHT)

        # الرسوم المتحركة
        self.play(Create(axes))
        self.play(Create(quadratic_func), run_time=3)
        self.wait(1)
        self.play(Create(vertex), Write(vertex_label))
        self.wait(1)
        self.play(Write(decreasing_text), Write(increasing_text))
        self.wait(2)
        
        # إظهار التمثيل البياني للدالة المشتقة
        derivative = axes.plot(
            lambda x: 2*x - 2,
            color=PURPLE,
            stroke_width=3
        )
        derivative_label = MathTex("f'(x) = 2x - 2", color=PURPLE)
            .to_edge(UP)
        
        self.play(Create(derivative), Write(derivative_label))
        self.wait(3)

هذا الكود ينتج فيديو متحرك يشرح دراسة الدالة التربيعية وتحليل اتجاه تغيراتها باستخدام المشتقة الأولى.

كود مانيم: تحويلات الدوال الرياضية

# تحويلات الدوال الرياضية باستخدام مانيم

from manim import *

class FunctionTransformations(Scene):
    def construct(self):
        axes = Axes(
            x_range=[-3, 3, 1],
            y_range=[-2, 2, 1],
            x_length=7,
            y_length=5
        )

        # الدالة الأساسية
        base_func = axes.plot(lambda x: np.sin(x), color=BLUE)
        base_label = MathTex("f(x) = \sin(x)").next_to(base_func, UP)

        # تحويلات الدالة
        transformed_func = axes.plot(lambda x: 2*np.sin(2*x + 1) + 0.5, color=RED)
        transformed_label = MathTex("g(x) = 2\sin(2x + 1) + 0.5").next_to(transformed_func, UP)

        self.play(Create(axes))
        self.play(Create(base_func), Write(base_label))
        self.wait(1)
        self.play(Transform(base_func, transformed_func),
                Transform(base_label, transformed_label))
        self.wait(2)

كود جيوجيبرا: إنشاء هندسي تفاعلي

// إنشاء مثلث ودائرة محيطة به باستخدام الجيوجيبرا

// إنشاء رؤوس المثلث
A = Point({0, 0})
B = Point({4, 0})
C = Point({2, 3})

// إنشاء المثلث
tri = Polygon(A, B, C)

// إنشاء منتصفات الأضلاع
mid_AB = Midpoint(A, B)
mid_BC = Midpoint(B, C)
mid_CA = Midpoint(C, A)

// إنشاء متوسطات المثلث
median_A = Segment(A, mid_BC)
median_B = Segment(B, mid_CA)
median_C = Segment(C, mid_AB)

// إيجاد نقطة تقاطع المتوسطات (مركز الثقل)
centroid = Intersect(median_A, median_B)

// تسمية النقاط
SetLabel(A, "A")
SetLabel(B, "B")
SetLabel(C, "C")
SetLabel(centroid, "G")

// تغيير ألوان العناصر
SetColor(tri, "blue")
SetColor(median_A, "red")
SetColor(median_B, "red")
SetColor(median_C, "red")
SetColor(centroid, "green")

# معلومات المشروع على GitHub
# العنوان: دراسة دالة تربيعية واتجاه تغيراتها بيانيا
# الرابط: https://github.com/beneddine-dotcom/manim_project/issues/24

## وصف المشروع
هذا المشروع يهدف إلى إنشاء فيديو تعليمي يشرح دراسة الدالة التربيعية:
f(x) = ax² + bx + c

## المميزات
1. إيجاد رأس المنحنى (الحد الأقصى/الأدنى)
2. دراسة اتجاه تغيرات الدالة (تزايدية/تناقصية)
3. إيجاد نقاط التقاطع مع المحاور
4. تمثيل الدالة المشتقة وتحليلها
